from copy import deepcopy
from typing import Dict

from sbmlsim.plot import Axis, Figure, Plot
from sbmlsim.simulation import Timecourse, TimecourseSim

from pkdb_models.models.canagliflozin.experiments.base_experiment import (
    CanagliflozinSimulationExperiment,
)
from pkdb_models.models.canagliflozin.helpers import run_experiments


class DoseDependencyExperiment(CanagliflozinSimulationExperiment):
    """Tests po application."""

    doses = [0, 2.5, 5, 10, 50, 100, 300]  # [mg]
    glucoses = [5, 6, 7, 8, 9, 10, 11]  # [mM]

    colors = ["black", "tab:orange", "tab:brown", "tab:purple", "tab:blue", "tab:red", "tab:green"]

    def simulations(self) -> Dict[str, TimecourseSim]:
        Q_ = self.Q_
        tcsims = {}

        for dose in self.doses:
            tcsims[f"can_dose_{dose}"] = TimecourseSim(
                Timecourse(
                    start=0,
                    end=24 * 60,  # [min]
                    steps=300,
                    changes={
                        **self.default_changes(),
                        f"PODOSE_can": Q_(dose, "mg"),
                    },
                )
            )

        for glc in self.glucoses:
            tcsims[f"can_glucose_{glc}"] = TimecourseSim(
                Timecourse(
                    start=0,
                    end=24 * 60,  # [min]
                    steps=300,
                    changes={
                        **self.default_changes(),
                        f"PODOSE_can": Q_(50, "mg"),
                        f"[KI__fpg]": Q_(glc, "mM"),  # FPG
                    },
                )
            )

        return tcsims

    def figures(self) -> Dict[str, Figure]:
        return {
            **self.figure_pk(),
            **self.figure_pd(),
        }

    def figure_pk(self) -> Dict[str, Figure]:
        figures = {}
        for key in ["dose", "glucose"]:
            if key == "dose":
                values = self.doses
            elif key == "glucose":
                values = self.glucoses
            fig = Figure(
                experiment=self,
                sid=f"Fig_{key}_dependency_pk",
                num_rows=3,
                num_cols=4,
            )
            plots = fig.create_plots(xaxis=Axis("time", unit="hr"), legend=True)
            sids = [
                # plasma
                "[Cve_can]",
                "[Cve_m5]",
                "[Cve_m7]",
                "[Cve_cantot]",
                # urine
                "Aurine_can",
                "Aurine_m5",
                "Aurine_m7",
                "Aurine_cantot",
                # feces
                "Afeces_can",
                "Afeces_m9",
                "Afeces_m7",
                "Afeces_cantot",
            ]
            for ksid, sid in enumerate(sids):
                plots[ksid].set_yaxis(label=self.labels[sid], unit=self.units[sid])

            for ksid, sid in enumerate(sids):
                for kval, value in enumerate(values):
                    plots[ksid].add_data(
                        task=f"task_can_{key}_{value}",
                        xid="time",
                        yid=sid,
                        label=f"{value}",
                        color=self.colors[kval],
                    )

            figures[fig.sid] = fig
        return figures

    def figure_pd(self) -> Dict[str, Figure]:
        figures = {}
        for key in ["dose", "glucose"]:
            if key == "dose":
                values = self.doses
            elif key == "glucose":
                values = self.glucoses

            fig = Figure(
                experiment=self,
                sid=f"Fig_{key}_dependency_pd",
                num_rows=2,
                num_cols=4,
            )
            plots = fig.create_plots(xaxis=Axis("time", unit="hr"), legend=True)
            sids = [
                "KI__UGE",
                "KI__RTG",
                "[KI__fpg]",
            ]
            for ksid, sid in enumerate(sids):
                plots[ksid].set_yaxis(label=self.labels[sid], unit=self.units[sid])

            for ksid, sid in enumerate(sids):
                for kval, value in enumerate(values):
                    # simulations
                    plots[ksid].add_data(
                        task=f"task_can_{key}_{value}",
                        xid="time",
                        yid=sid,
                        label=f"{value}",
                        color=self.colors[kval],
                    )

            for ksid in range(4, 8):
                plots[ksid].set_xaxis(label=self.label_can, unit=self.unit_can)

            plots[4].set_yaxis(label="Rate glucose excretion", unit="mmole/min")
            plots[5].set_yaxis(label=self.label_rtg, unit=self.unit_rtg)
            plots[6].set_yaxis(label=self.label_uge, unit=self.unit_uge)
            for kval, value in enumerate(values):
                # simulations
                plots[4].add_data(
                    task=f"task_can_{key}_{value}",
                    xid="[Cve_can]",
                    yid="KI__GLCEX",
                    label=f"{value} mg",
                    color=self.colors[kval],
                )
                plots[5].add_data(
                    task=f"task_can_{key}_{value}",
                    xid="[Cve_can]",
                    yid="KI__RTG",
                    label=f"{value} mg",
                    color=self.colors[kval],
                )
                plots[6].add_data(
                    task=f"task_can_{key}_{value}",
                    xid="[Cve_can]",
                    yid="KI__UGE",
                    label=f"{value} mg",
                    color=self.colors[kval],
                )

            figures[fig.sid] = fig
        return figures


if __name__ == "__main__":
