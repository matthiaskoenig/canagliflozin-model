"""Parameter scans canagliflozin."""
from pathlib import Path
from typing import Dict

import matplotlib.axes
import matplotlib.cm as cm

import numpy as np
from pint import UnitRegistry
from sbmlsim.simulation import Timecourse, TimecourseSim, ScanSim, Dimension
from sbmlsim.plot.serialization_matplotlib import FigureMPL, MatplotlibFigureSerializer
from sbmlsim.plot.serialization_matplotlib import plt
from sbmlutils.console import console

from pkdb_models.models.canagliflozin.experiments.base_experiment import (
    CanagliflozinSimulationExperiment,
)
from pkdb_models.models.canagliflozin.helpers import run_experiments


class CanagliflozinParameterScan(CanagliflozinSimulationExperiment):
    """Scan the effect of parameters on pharmacokinetics."""

    num_points = 11  # 19
    scan_map = {

        "renal_scan": {
            "parameter": "KI__f_renal_function",
            "default": 1.0,
            "range": np.sort(
                np.append(np.logspace(-1, 1, num=num_points), [1.0])
            ),  # [10^-1=0.1, 10^1=10]
            "scale": "log",
            "colormap": "seismic_r",
            "units": "dimensionless",
            "label": "renal function [-]",
        },
        #
        # "hepatic_scan": {
        #     "parameter": "LI__f_hepatic_function",
        #     "default": 1.0,
        #     "range": np.sort(
        #         np.append(np.logspace(-1, 1, num=num_points), [1.0])
        #     ),  # [10^-1=0.1, 10^1=10]
        #     "scale": "log",
        #     "colormap": "seismic_r",
        #     "units": "dimensionless",
        #     "label": "hepatic function [-]",
        # },

        "hepatic_scan": {
            "parameter": "f_cirrhosis",
            "default": 0.0,
            "range": np.linspace(0, 0.9, num=num_points),
            # "range": np.logspace(-2, 2, num=21),
            "scale": "linear",
            "colormap": "Reds",
            "units": "dimensionless",
            "label": "Cirrhosis degree [-]",
        },

    }

    def simulations(self) -> Dict[str, ScanSim]:
        Q_ = self.Q_
        tcscans = {}

        for scan_key, scan_data in self.scan_map.items():

            # renal simulations
            for renal_key in self.renal_map.keys():
                tcscans[f"scan_po_{scan_key}_{renal_key}"] = ScanSim(
                    simulation=TimecourseSim(
                        Timecourse(
                            start=0,
                            end=36 * 60,
                            steps=2000,
                            changes={
                                **self.default_changes(),
                                "PODOSE_can": Q_(100, "mg"),
                                "KI__f_renal_function": Q_(self.renal_map[renal_key], "dimensionless")
                            },
                        )
                    ),
                    dimensions=[
                        Dimension(
                            "dim_scan",
                            changes={
                                scan_data["parameter"]: Q_(
                                    scan_data["range"], scan_data["units"]
                                )
                            },
                        ),
                    ],
                )

            # hepatic simulations
            for hepatic_key in self.cirrhosis_map.keys():
                tcscans[f"scan_po_{scan_key}_{hepatic_key}"] = ScanSim(
                    simulation=TimecourseSim(
                        Timecourse(
                            start=0,
                            end=36 * 60,
                            steps=2000,
                            changes={
                                **self.default_changes(),
                                "PODOSE_can": Q_(100, "mg"),
                                "f_cirrhosis": Q_(self.cirrhosis_map[hepatic_key], "dimensionless")
                            },
                        )
                    ),
                    dimensions=[
                        Dimension(
                            "dim_scan",
                            changes={
                                scan_data["parameter"]: Q_(
                                    scan_data["range"], scan_data["units"]
                                )
                            },
                        ),
                    ],
                )

        return tcscans

    def figures_mpl(self) -> Dict[str, FigureMPL]:
        """Matplotlib figures."""
        # calculate pharmacokinetic parameters
        self.pk_dfs = self.calculate_canagliflozin_pk()
        # console.print(self.pk_dfs)

        return {
            **self.figures_mpl_timecourses(),
            **self.figures_mpl_pharmacokinetics(),
        }

    def figures_mpl_timecourses(self) -> Dict[str, FigureMPL]:
        """Timecourse plots for key variables depending on degree of renal impairment."""

        flip: bool = False  # flip rows and columns
        sids = [
            "[Cve_can]",
            "Aurine_can",
            "Afeces_can",

            "[Cve_m5]",
            # "Aurine_m5",

            "[Cve_m7]",
            # "Aurine_m7",
            # "Afeces_m7",

            # "[Cve_cantot]",
            # "Aurine_cantot",
        ]

        figures = {}

        for scan_key, scan_data in self.scan_map.items():
            range = scan_data["range"]
            rmin, rmax = range[0], range[-1]

            # cmap_str
            cmap_str = scan_data["colormap"]
            cmap = matplotlib.colormaps.get_cmap(cmap_str)

            for disease_type in ["renal", "hepatic"]:
                if disease_type == "renal" and scan_key == "renal_scan":
                    continue
                if disease_type == "hepatic" and scan_key == "hepatic_scan":
                    continue

                disease_map = self.renal_map if disease_type == "renal" else self.cirrhosis_map

                nrows = len(sids)
                ncols = len(disease_map)
                if flip:
                    nrows, ncols = ncols, nrows
                figsize = (6 * ncols, 6 * nrows)  # (width, height)

                f, axes = plt.subplots(
                    nrows=nrows,
                    ncols=ncols,
                    figsize=figsize,
                    sharey="row",
                    sharex="col",
                    dpi=150,
                    layout="constrained",
                    squeeze=False,
                )

                ymax = {}
                for k_sid, sid in enumerate(sids):
                    ymax[sid] = 0.0
                    for k_disease, disease_key in enumerate(disease_map.keys()):
                        krow, kcol = k_sid, k_disease
                        if flip:
                            krow, kcol = k_disease, k_sid

                        ax = axes[krow, kcol]

                        # get data
                        Q_ = self.Q_
                        xres = self.results[
                            f"task_scan_po_{scan_key}_{disease_key}"
                        ]

                        # scanned dimension
                        scandim = xres._redop_dims()[0]
                        parameter_id = scan_data["parameter"]
                        par_vec = Q_(
                            xres[parameter_id].values[0], xres.uinfo[parameter_id]
                        )
                        t_vec = xres.dim_mean("time").to(self.units["time"])

                        for k_par, par in enumerate(par_vec):
                            c_vec = Q_(
                                xres[sid].sel({scandim: k_par}).values,
                                xres.uinfo[sid],
                            ).to(self.units[sid])

                            # update ymax
                            cmax = np.nanmax(c_vec.magnitude)
                            if cmax > ymax[sid]:
                                ymax[sid] = cmax

                            # plot all curves for the scan
                            facecolors = self.cirrhosis_colors if disease_type == "hepatic" else self.renal_colors
                            facecolor = facecolors[disease_key]
                            if facecolor != "black":
                                for axis in ["top", "bottom", "left", "right"]:
                                    ax.spines[axis].set_linewidth(4)
                                    ax.spines[axis].set_color(facecolor)

                            # 0.1 - 1.9
                            linewidth = 2.0
                            if np.isclose(scan_data["default"], par):
                                color = "black"
                                t_vec_default = t_vec
                                c_vec_default = c_vec
                            else:
                                # red less function, blue more function
                                if scan_data["scale"] == "linear":
                                    cvalue = (par-rmin)/np.abs(rmax-rmin)
                                elif scan_data["scale"] == "log":
                                    cvalue = (np.log10(par) - np.log10(rmin)) / np.abs(np.log10(rmax) - np.log10(rmin))

                                color = cmap(cvalue.magnitude)

                            ax.plot(
                                t_vec.magnitude,
                                c_vec.magnitude,
                                color=color,
                                linewidth=linewidth,
                            )

                        # plot the reference line in black
                        ax.plot(
                            t_vec_default.magnitude,
                            c_vec_default.magnitude,
                            color="black",
                            linewidth=2.0,
                        )

                        ax: matplotlib.axes.Axes
                        if krow == 0 and not flip:
                            ax.set_title(
                                disease_key,
                                fontdict={
                                    "fontsize": self.tick_font_size,
                                    "fontweight": "bold",
                                },
                            )

                        if krow == nrows-1:
                            ax.set_xlabel(
                                f"{self.label_time} [{self.units['time']}]",
                                fontdict=self.font,
                            )
                        ax.tick_params(axis="x", labelsize=self.tick_font_size)
                        ax.tick_params(axis="y", labelsize=self.tick_font_size)

                        ax.set_ylabel(
                            f"{self.labels[sid]} [{self.units[sid]}]",
                            fontdict=self.font,
                        )

                # adjust ylimits equal for all plots
                for krow, sid in enumerate(sids):
                    for kcol, disease_key in enumerate(disease_map.keys()):
                        if not flip:
                            ax = axes[krow, kcol]
                        else:
                            ax = axes[kcol, krow]
                        ax.set_ylim(bottom=0.0, top=1.05 * ymax[sid])
                        ax.set_xlim(right=36)


                # --- colorbar ---
                # 4-tuple of floats rect = (left, bottom, width, height).
                # A new Axes is added with dimensions rect in normalized (0, 1)

                # use figsize to add the colorbar at the right position

                # rect : This parameter is the dimensions [left, bottom, width, height] of the new axes.

                # add space
                engine = f.get_layout_engine()
                engine.set(rect=(0.0, 0.7, 0.0, 0.0))
                cb_ax = f.add_axes(rect=[0.2, 0.98, 0.6, 0.04/nrows])
                cb_ax.set_in_layout(True)

                # colorbar range
                if scan_data["scale"] == "linear":
                    norm = matplotlib.colors.Normalize(vmin=rmin, vmax=rmax, clip=False)
                elif scan_data["scale"] == "log":
                    norm = matplotlib.colors.LogNorm(vmin=rmin, vmax=rmax, clip=False)

                cbar = f.colorbar(
                    cm.ScalarMappable(norm=norm, cmap=cmap_str),
                    cax=cb_ax,
                    orientation="horizontal",
                )

                # ticks
                ticks = [rmin, rmax]
                if scan_data["default"] not in ticks:
                    ticks.append(scan_data["default"])
                    ticks = sorted(ticks)
                cbar.set_ticks(ticks)

                cbar.set_ticklabels(
                    ticks, **{"size": 15, "weight": "medium"}
                )
                cbar.ax.set_xlabel(
                    scan_data["label"], **{"size": 15, "weight": "bold"}
                )
                cbar.ax.axvline(x=scan_data["default"], color="black", linewidth=2)

                figures[f"fig_can_po__{disease_type}__{scan_key}__timecourse"] = f


        return figures

    def figures_mpl_pharmacokinetics(self):
        """Visualize dependency of pharmacokinetics parameters."""
        Q_ = self.Q_
        figures = {}

        parameters_info = {
            # ["canagliflozin", "M5", "M7", "M9", "total canagliflozin"]
            "canagliflozin": [
                "aucinf",
                "kel",
                "vd",
                "cmax",
                "thalf",
                "cl",
                # "cl_renal",
                # "cl_fecal",
            ],
            "M5": [
                "aucinf",
                "thalf",
            ],
            "M7":
            [
                "aucinf",
                "thalf",
            ],
            # "total canagliflozin":
            # [
            #     "aucinf",
            #     "thalf",
            # ],
            # "M9":
            # [
            #     "aucinf",
            #     "thalf",
            # ]
        }

        for substance, parameters in parameters_info.items():
            for scan_key, scan_data in self.scan_map.items():
                for disease_type in ["renal", "hepatic"]:
                    if disease_type == "renal" and scan_key == "renal_scan":
                        continue
                    if disease_type == "hepatic" and scan_key == "hepatic_scan":
                        continue

                    disease_map = self.renal_map if disease_type == "renal" else self.cirrhosis_map

                    f, axes = plt.subplots(
                        nrows=1, ncols=len(parameters), figsize=(6 * len(parameters), 6), dpi=150,
                        layout="constrained",
                        squeeze = False,
                    )
                    f.suptitle(
                        f"Pharmacokinetic parameters {substance}",
                        fontsize=self.suptitle_font_size,
                    )
                    axes = axes.flatten()

                    for k, pk_key in enumerate(parameters):
                        ax = axes[k]
                        ax.axvline(x=scan_data["default"], color="grey", linestyle="--")

                        ymax = 0.0
                        for disease_key in disease_map:

                            sim_key = f"scan_po_{scan_key}_{disease_key}"
                            xres = self.results[f"task_{sim_key}"]
                            df = self.pk_dfs[sim_key]
                            df = df[df.substance == substance]  # get PK for substance

                            # This was scanned
                            parameter_id = scan_data["parameter"]
                            x_vec = Q_(
                                xres[parameter_id].values[0], xres.uinfo[parameter_id]
                            )

                            pk_vec = df[f"{pk_key}"]
                            pk_vec = pk_vec.to_numpy()

                            x = x_vec
                            y = Q_(pk_vec, df[f"{pk_key}_unit"].values[0])

                            y = y.to(self.pk_units[pk_key])
                            colors = self.renal_colors if disease_type == "renal" else self.cirrhosis_colors
                            ax.plot(
                                x,
                                y,
                                marker="o",
                                linestyle="-",
                                color=colors[disease_key],
                                markeredgecolor="black",
                                label=disease_key,
                            )
                            ymax_value = np.nanmax(y.magnitude)
                            if ymax_value > ymax:
                                ymax = ymax_value

                        ax.tick_params(
                            axis="x", labelsize=self.tick_font_size
                        )
                        ax.tick_params(
                            axis="y", labelsize=self.tick_font_size
                        )
                        # ax.set_xlabel(scan_data["label"], fontdict=CanagliflozinSimulationExperiment.scan_font)
                        ax.set_xlabel(
                            scan_data["label"],
                            fontdict=self.scan_font,
                        )
                        ax.set_ylabel(
                            f"{self.pk_labels[pk_key]} ({substance}) [{self.pk_units[pk_key]}]",
                            fontdict=self.scan_font,
                        )

                        # set axis
                        ax.set_ylim(bottom=0.0, top=1.05 * ymax)
                        ax.set_xscale("log")
                        # ax.legend(fontsize=CanagliflozinSimulationExperiment.legend_font_size)
                        # ax.legend()

                    figures[f"fig_can_po__{substance}__{disease_type}__{scan_key}__parameters"] = f
        return figures
